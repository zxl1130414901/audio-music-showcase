# 1) 分析：onset / tempo / pitch，切片与每片音高
$env:TMP='F:\coding音乐\_env\tmp'; $env:TEMP='F:\coding音乐\_env\tmp'

F:\coding音乐\_env\venv_vts\Scripts\python.exe F:\coding音乐\_tmp\C1\aubio\render.py

F:\coding音乐\_env\venv_vts\Scripts\python.exe F:\coding音乐\_tmp\C1\aubio\check.py
powershell -NoProfile -ExecutionPolicy Bypass -File "F:\coding音乐\_env\verify_audio.ps1" -Path "F:\coding音乐\_samples\aubio__aubio\sample.wav"

# 4) 波形/频谱图（截取 1-19 s，避开片尾淡出）
C:\ffmpeg\bin\ffmpeg.exe -y -ss 1 -t 18 -i sample.wav -filter_complex "showwavespic=s=1280x360:colors=0xffc247" -frames:v 1 waveform.png
C:\ffmpeg\bin\ffmpeg.exe -y -ss 1 -t 18 -i sample.wav -lavfi "showspectrumpic=s=1280x640:legend=1" -frames:v 1 spectrogram.png
